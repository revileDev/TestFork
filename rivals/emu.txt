local Emulator = {}

local UNIVERSE_ID = 6035872082
local CALLBACK_TIMEOUT = 5
local MATERIAL_INTERVAL = 4
local LOG_LEVELS = { DEBUG = 1, INFO = 2, WARN = 3, ERROR = 4 }

local state = {
    started = false,
    active = false,
    startedAt = nil,
    challenge = nil,
    native = nil,
    originalScript = nil,
    inbound = 0,
    invoked = 0,
    responses = 0,
    malformed = 0,
    errors = 0,
    materialSent = 0,
    materialErrors = 0,
    miscFound = 0,
    miscCancelled = 0,
    miscFailed = 0,
    lastError = nil,
    lastChallengeAt = nil,
    lastResponseAt = nil,
    logLevel = LOG_LEVELS.DEBUG,
}

local function log(level, event, message)
    if LOG_LEVELS[level] < state.logLevel then return end
    local line = string.format("[Emulator][%s][%s] %s", level, event, message)
    if level == "WARN" or level == "ERROR" then
        warn(line)
    else
        print(line)
    end
end

local function traceback(problem)
    return debug.traceback(tostring(problem), 2)
end

local function capture(callback, ...)
    local ok, result = xpcall(callback, traceback, ...)
    return if ok then nil else result
end

local function sourceMatches(callback, needle)
    local ok, source = pcall(debug.info, callback, "s")
    return ok and type(source) == "string"
        and source:gsub("%s+", ""):find(needle, 1, true) ~= nil
end

local function findChallenge(remotes)
    local deadline = os.clock() + CALLBACK_TIMEOUT
    repeat
        local challenge, native
        for _, candidate in ipairs(remotes:QueryDescendants("RemoteEvent")) do
            for _, connection in ipairs(getconnections(candidate.OnClientEvent)) do
                local callback = connection.Function
                if type(callback) == "function"
                    and sourceMatches(callback, "ReplicatedFirst.LocalScript3") then
                    assert(not native, "ambiguous LocalScript3 challenge callback")
                    challenge, native = candidate, connection
                end
            end
        end
        if native then return challenge, native end
        task.wait(0.1)
    until os.clock() >= deadline
    error("LocalScript3 challenge callback missing; rejoin and retry")
end

local function observeResponses(challenge)
    local previousNamecall
    previousNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        if state.active and self == challenge and getnamecallmethod() == "FireServer" then
            local args = table.pack(...)
            local packet, validation = args[1], args[3]
            if args.n == 5 and type(packet) == "table"
                and type(packet[#packet]) == "table"
                and type(validation) == "table" and #validation == 3 then
                state.responses += 1
                state.lastResponseAt = os.clock()
                local latency = state.lastChallengeAt
                    and math.floor((state.lastResponseAt - state.lastChallengeAt) * 1000 + 0.5)
                    or -1
                log("DEBUG", "CHALLENGE_RESPONSE", string.format(
                    "count=%d latencyMs=%d packetItems=%d",
                    state.responses, latency, #packet
                ))
            else
                state.malformed += 1
                if state.malformed <= 3 or state.malformed % 10 == 0 then
                    log("WARN", "CHALLENGE_SHAPE", string.format(
                        "unexpected response shape; count=%d", state.malformed
                    ))
                end
            end
        end
        return previousNamecall(self, ...)
    end))
end

local function attachRelay(challenge, native)
    local solve = native.Function
    local relay
    relay = challenge.OnClientEvent:Connect(function(...)
        state.inbound += 1
        state.lastChallengeAt = os.clock()
        local args = table.pack(...)
        local failure = capture(function()
            solve(table.unpack(args, 1, args.n))
        end)
        if failure then
            state.errors += 1
            state.lastError = failure
            log("ERROR", "CHALLENGE_CALLBACK", failure)
            state.active = false
            relay:Disconnect()
            native:Enable()
            state.originalScript.Enabled = true
            log("WARN", "RELAY_FALLBACK", "native connection restored after callback failure")
        else
            state.invoked += 1
        end
    end)

    native:Disable()
    if native.Enabled then
        relay:Disconnect()
        error("could not detach original challenge connection")
    end
    state.originalScript.Enabled = false
    assert(not state.originalScript.Enabled, "LocalScript3 remained enabled")
    state.active = true
end

local function startMaterialHeartbeat(materialRemote)
    task.spawn(function()
        while state.active do
            local failure = capture(function()
                materialRemote:FireServer("", nil, nil)
            end)
            if failure then
                state.materialErrors += 1
                state.lastError = failure
                if state.materialErrors <= 3 or state.materialErrors % 10 == 0 then
                    log("ERROR", "MATERIAL_SEND", failure)
                end
            else
                state.materialSent += 1
            end
            task.wait(MATERIAL_INTERVAL)
        end
    end)
end

local function belongsToMisc(thread)
    for level = 1, 24 do
        local ok, source = pcall(debug.info, thread, level, "s")
        if not ok or not source then break end
        if type(source) == "string"
            and source:gsub("%s+", ""):find("MiscellaneousController", 1, true) then
            return true
        end
    end
    return false
end

local function cancelMiscThreads()
    local targets = {}
    for _, thread in ipairs(getgc(true)) do
        if type(thread) == "thread" and coroutine.status(thread) == "suspended"
            and belongsToMisc(thread) then
            targets[#targets + 1] = thread
        end
    end

    state.miscFound = #targets
    for _, thread in ipairs(targets) do
        local ok = pcall(task.cancel, thread)
        if not ok then ok = pcall(coroutine.close, thread) end
        if ok then
            state.miscCancelled += 1
        else
            state.miscFailed += 1
        end
    end
    if state.miscFailed > 0 then
        log("WARN", "MISC_TASKS", string.format(
            "cancelFailed=%d", state.miscFailed
        ))
    end
end

function Emulator.setLogLevel(level)
    assert(LOG_LEVELS[level], "invalid log level")
    state.logLevel = LOG_LEVELS[level]
end

function Emulator.status()
    local now = os.clock()
    return {
        started = state.started,
        active = state.active,
        uptime = state.startedAt and now - state.startedAt or nil,
        remote = state.challenge and state.challenge:GetFullName() or nil,
        originalScriptEnabled = if state.originalScript then state.originalScript.Enabled else nil,
        originalConnectionEnabled = if state.native then state.native.Enabled else nil,
        inbound = state.inbound,
        invoked = state.invoked,
        responses = state.responses,
        malformed = state.malformed,
        errors = state.errors,
        materialSent = state.materialSent,
        materialErrors = state.materialErrors,
        miscFound = state.miscFound,
        miscCancelled = state.miscCancelled,
        miscFailed = state.miscFailed,
        lastError = state.lastError,
        challengeAge = state.lastChallengeAt and now - state.lastChallengeAt or nil,
        responseAge = state.lastResponseAt and now - state.lastResponseAt or nil,
    }
end

function Emulator.start()
    if state.started then return Emulator end
    assert(game.GameId == UNIVERSE_ID, "RIVALS universe expected")
    assert(type(getconnections) == "function", "getconnections unavailable")
    assert(type(hookmetamethod) == "function", "hookmetamethod unavailable")
    assert(type(newcclosure) == "function", "newcclosure unavailable")
    assert(type(getgc) == "function", "getgc unavailable")

    local replicatedFirst = game:GetService("ReplicatedFirst")
    local originalScript = replicatedFirst:WaitForChild("LocalScript3")
    if not originalScript.Enabled then originalScript.Enabled = true end
    local remotes = game:GetService("ReplicatedStorage"):WaitForChild("Remotes")
    local materialRemote = game:GetService("MaterialService"):WaitForChild("RemoteEvent")
    local challenge, native = findChallenge(remotes)
    if not native.Enabled then native:Enable() end
    assert(native.Enabled, "challenge callback could not be restored")

    state.originalScript = originalScript
    state.challenge = challenge
    state.native = native
    observeResponses(challenge)
    attachRelay(challenge, native)
    startMaterialHeartbeat(materialRemote)
    cancelMiscThreads()
    state.started = true
    state.startedAt = os.clock()
    log("INFO", "STARTED", "Emulator started")
    return Emulator
end

Emulator.start()
return table.freeze(Emulator)
